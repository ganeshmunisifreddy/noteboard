**Goal:** Notes and lists get one date (optional time, optional repeat), reminders while Noteboard is open, Mark done, and an Upcoming agenda.

```ts
// Calendar dates as 'YYYY-MM-DD' strings. Day and month arithmetic runs in
// UTC, so a clock change never adds or loses a day; only localDate and
// localMoment use the device's time zone.

export type DateParts = { year: number; month: number; day: number }; // month 1–12

export const WEEKDAY_SHORT = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
export const WEEKDAY_LONG = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
export const MONTH_SHORT = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

const DAY_MS = 24 * 60 * 60 * 1000;
const DATE_PATTERN = /^(\d{4})-(\d{2})-(\d{2})$/;
const TIME_PATTERN = /^([01]\d|2[0-3]):[0-5]\d$/;

export const daysInMonth = (year: number, month: number) => new Date(Date.UTC(year, month, 0)).getUTCDate();

export const parseDate = (date: string): DateParts => {
  const [year, month, day] = date.split('-').map(Number);
  return { year, month, day };
};

const pad = (value: number, length = 2) => String(value).padStart(length, '0');

export const formatDate = ({ year, month, day }: DateParts) => `${pad(year, 4)}-${pad(month)}-${pad(day)}`;

// Years from 1000 on: Date.UTC reads years below 100 as 19xx.
export const isDateString = (value: unknown): value is string => {
  if (typeof value !== 'string') return false;
  const match = DATE_PATTERN.exec(value);
  if (!match) return false;
  const [year, month, day] = match.slice(1).map(Number);
  return year >= 1000 && month >= 1 && month <= 12 && day >= 1 && day <= daysInMonth(year, month);
};

export const isTimeString = (value: unknown): value is string => typeof value === 'string' && TIME_PATTERN.test(value);

const dayNumber = (date: string) => {
  const { year, month, day } = parseDate(date);
  return Math.round(Date.UTC(year, month - 1, day) / DAY_MS);
};

const fromDayNumber = (days: number) => {
  const d = new Date(days * DAY_MS);
  return formatDate({ year: d.getUTCFullYear(), month: d.getUTCMonth() + 1, day: d.getUTCDate() });
};

export const addDays = (date: string, days: number) => fromDayNumber(dayNumber(date) + days);

export const daysBetween = (from: string, to: string) => dayNumber(to) - dayNumber(from);

export const weekdayOf = (date: string) => new Date(dayNumber(date) * DAY_MS).getUTCDay();

// The same day of the month `months` later, or that month's last day when
// it's shorter (31 Jan + 1 month = 28 Feb).
export const addMonths = (date: string, months: number) => {
  const { year, month, day } = parseDate(date);
  const index = year * 12 + (month - 1) + months;
  const nextYear = Math.floor(index / 12);
  const nextMonth = (index % 12) + 1;
  return formatDate({ year: nextYear, month: nextMonth, day: Math.min(day, daysInMonth(nextYear, nextMonth)) });
};

export const monthsBetween = (from: string, to: string) => {
  const a = parseDate(from);
  const b = parseDate(to);
  return (b.year - a.year) * 12 + (b.month - a.month);
};

// The device's local date at a moment.
export const localDate = (ms: number) => {
  const d = new Date(ms);
  return formatDate({ year: d.getFullYear(), month: d.getMonth() + 1, day: d.getDate() });
};

// A local date and time as ms since epoch.
export const localMoment = (date: string, time: string) => {
  const { year, month, day } = parseDate(date);
  const [hours, minutes] = time.split(':').map(Number);
  return new Date(year, month - 1, day, hours, minutes).getTime();
};

export const shortDate = (date: string) => {
  const { year, month, day } = parseDate(date);
  return `${day} ${MONTH_SHORT[month - 1]} ${year}`;
};
```

```ts
import { addDays, addMonths, daysBetween, monthsBetween, shortDate, WEEKDAY_SHORT, weekdayOf } from './calendar';
import type { Repeat, RepeatUnit } from './types';

export const REPEAT_UNITS: RepeatUnit[] = ['day', 'week', 'month', 'year'];
export const REPEAT_PRESETS = ['daily', 'weekdays', 'weekly', 'monthly', 'yearly'] as const;
export type RepeatPreset = (typeof REPEAT_PRESETS)[number];

const MON_TO_FRI = [1, 2, 3, 4, 5];

export const presetRepeat = (preset: RepeatPreset, anchor: string): Repeat => {
  const base: Repeat = { every: 1, unit: 'day', weekdays: [], anchor, ends: { type: 'never' } };
  switch (preset) {
    case 'daily':
      return base;
    case 'weekdays':
      return { ...base, unit: 'week', weekdays: MON_TO_FRI };
    case 'weekly':
      return { ...base, unit: 'week' };
    case 'monthly':
      return { ...base, unit: 'month' };
    case 'yearly':
      return { ...base, unit: 'year' };
  }
};

const sameDays = (a: number[], b: number[]) => {
  const x = [...a].sort();
  const y = [...b].sort();
  return x.length === y.length && x.every((day, i) => day === y[i]);
};

export const presetOf = (repeat: Repeat): RepeatPreset | 'custom' => {
  if (repeat.every !== 1 || repeat.ends.type !== 'never') return 'custom';
  switch (repeat.unit) {
    case 'day':
      return 'daily';
    case 'week':
      if (repeat.weekdays.length === 0) return 'weekly';
      return sameDays(repeat.weekdays, MON_TO_FRI) ? 'weekdays' : 'custom';
    case 'month':
      return 'monthly';
    case 'year':
      return 'yearly';
  }
};

// Weeks start on Monday.
const weekStart = (date: string) => addDays(date, -((weekdayOf(date) + 6) % 7));

const following = ({ anchor, every, unit, weekdays }: Repeat, after: string): string | null => {
  if (unit === 'week' && weekdays.length > 0) {
    // The chosen weekdays, in every `every`-th week counted from the anchor's week.
    const start = weekStart(anchor);
    let day = anchor > after ? anchor : addDays(after, 1);
    for (let i = 0; i < every * 7 + 7; i += 1, day = addDays(day, 1)) {
      const week = Math.floor(daysBetween(start, day) / 7);
      if (week % every === 0 && weekdays.includes(weekdayOf(day))) return day;
    }
    return null;
  }
  if (anchor > after) return anchor;
  if (unit === 'day' || unit === 'week') {
    const step = unit === 'day' ? every : every * 7;
    return addDays(anchor, (Math.floor(daysBetween(anchor, after) / step) + 1) * step);
  }
  // Months and years count from the anchor, so the 31st comes back after a short month.
  const step = unit === 'month' ? every : every * 12;
  let k = Math.max(1, Math.floor(monthsBetween(anchor, after) / step));
  while (addMonths(anchor, k * step) <= after) k += 1;
  return addMonths(anchor, k * step);
};

// The first date of the repeat after `after`, or null past its end date.
// "After N times" is counted by Mark done (lib/due.ts), not here.
export const nextDate = (repeat: Repeat, after: string): string | null => {
  const next = following(repeat, after);
  if (next === null) return null;
  if (repeat.ends.type === 'on' && next > repeat.ends.date) return null;
  return next;
};

const UNIT_PLURALS: Record<RepeatUnit, string> = { day: 'days', week: 'weeks', month: 'months', year: 'years' };

// Monday first, as in the picker.
const dayList = (days: number[]) =>
  [...days]
    .sort((a, b) => ((a + 6) % 7) - ((b + 6) % 7))
    .map((day) => WEEKDAY_SHORT[day])
    .join(', ');

const PRESET_NAMES: Record<RepeatPreset, string> = {
  daily: 'Daily',
  weekdays: 'Weekdays',
  weekly: 'Weekly',
  monthly: 'Monthly',
  yearly: 'Yearly',
};

export const repeatText = (repeat: Repeat): string => {
  const preset = presetOf({ ...repeat, ends: { type: 'never' } });
  const onDays = repeat.unit === 'week' && repeat.weekdays.length > 0 ? ` on ${dayList(repeat.weekdays)}` : '';
  let text: string;
  if (preset !== 'custom') text = PRESET_NAMES[preset];
  else if (repeat.every === 1) text = `Weekly${onDays}`;
  else text = `Every ${repeat.every} ${UNIT_PLURALS[repeat.unit]}${onDays}`;
  if (repeat.ends.type === 'on') text += `, until ${shortDate(repeat.ends.date)}`;
  if (repeat.ends.type === 'after') {
    const left = repeat.ends.count - repeat.ends.done;
    text += `, ${left} ${left === 1 ? 'time' : 'times'} left`;
  }
  return text;
};
```

```ts
import { addDays, localDate, localMoment, MONTH_SHORT, parseDate, WEEKDAY_SHORT, weekdayOf } from './calendar';
import { nextDate, repeatText } from './repeat';
import type { Due, Repeat } from './types';

export const DEFAULT_TIME = '09:00';
export const DONE_DAYS = 30;

const DAY_MS = 24 * 60 * 60 * 1000;

// A new or changed date starts fresh: not done, not alerted, not snoozed,
// and its repeat counts from this date.
export const newDue = (date: string, time: string | null, repeat: Repeat | null): Due => ({
  date,
  time,
  repeat: repeat && {
    ...repeat,
    anchor: date,
    ends: repeat.ends.type === 'after' ? { ...repeat.ends, done: 0 } : repeat.ends,
  },
  doneAt: null,
  alertedAt: null,
  snoozedUntil: null,
});

export const dueMoment = (due: Pick<Due, 'date' | 'time'>) => localMoment(due.date, due.time ?? DEFAULT_TIME);

export const alertTime = (due: Due) => due.snoozedUntil ?? dueMoment(due);

export const isDone = (due: Due) => due.doneAt !== null;

// With a time: once it has passed. Without: from the next day.
export const isOverdue = (due: Due, now: number) =>
  !isDone(due) && (due.time ? now > dueMoment(due) : localDate(now) > due.date);

const withDone = (repeat: Repeat, done: number): Repeat =>
  repeat.ends.type === 'after' ? { ...repeat, ends: { ...repeat.ends, done } } : repeat;

// Mark done. A date that doesn't repeat is done. A repeat moves to its next
// date, skipping missed ones to today or later; dates moved past count
// toward "after N times". Once the series ends the note is done on its last date.
export const completeDue = (due: Due, now: number): Due => {
  const cleared: Due = { ...due, alertedAt: null, snoozedUntil: null };
  const { repeat } = due;
  if (!repeat) return { ...cleared, doneAt: now };
  const today = localDate(now);
  if (repeat.ends.type === 'never') {
    let next = nextDate(repeat, due.date);
    if (next !== null && next < today) next = nextDate(repeat, addDays(today, -1));
    return next === null ? { ...cleared, doneAt: now } : { ...cleared, date: next, doneAt: null };
  }
  let date = due.date;
  let done = repeat.ends.type === 'after' ? repeat.ends.done : 0;
  for (;;) {
    const ended = repeat.ends.type === 'after' && done + 1 >= repeat.ends.count;
    const next = ended ? null : nextDate(repeat, date);
    if (next === null) return { ...cleared, date, repeat: withDone(repeat, done), doneAt: now };
    date = next;
    done += 1;
    if (date >= today) return { ...cleared, date, repeat: withDone(repeat, done), doneAt: null };
  }
};

// The browser's own 12- or 24-hour style; any odd spaces Intl uses become plain spaces.
export const formatTime = (time: string, locale?: string) => {
  const [hours, minutes] = time.split(':').map(Number);
  return new Intl.DateTimeFormat(locale, { hour: 'numeric', minute: '2-digit' })
    .format(new Date(2000, 0, 1, hours, minutes))
    .replace(/\s/g, ' ');
};

const dayAndMonth = (date: string) => {
  const { month, day } = parseDate(date);
  return `${WEEKDAY_SHORT[weekdayOf(date)]} ${day} ${MONTH_SHORT[month - 1]}`;
};

export const formatDay = (date: string, today: string) => {
  if (date === today) return 'Today';
  if (date === addDays(today, 1)) return 'Tomorrow';
  if (date === addDays(today, -1)) return 'Yesterday';
  const text = dayAndMonth(date);
  return parseDate(date).year === parseDate(today).year ? text : `${text} ${parseDate(date).year}`;
};

export const formatDue = (due: Pick<Due, 'date' | 'time'>, now: number, locale?: string) => {
  const day = formatDay(due.date, localDate(now));
  return due.time ? `${day}, ${formatTime(due.time, locale)}` : day;
};

// Always with the year, for the Markdown export.
export const formatDueLong = (due: Pick<Due, 'date' | 'time'>, locale?: string) => {
  const day = `${dayAndMonth(due.date)} ${parseDate(due.date).year}`;
  return due.time ? `${day}, ${formatTime(due.time, locale)}` : day;
};

export const lowerFirst = (text: string) => text.charAt(0).toLowerCase() + text.slice(1);

// e.g. "Due today, 6:00 PM, overdue, repeats weekly"
export const dueDescription = (due: Due, now: number, locale?: string) => {
  const when = formatDue(due, now, locale).replace(/^(Today|Tomorrow|Yesterday)/, (word) => word.toLowerCase());
  const parts = [isDone(due) ? `Done, was due ${when}` : `Due ${when}`];
  if (isOverdue(due, now)) parts.push('overdue');
  if (due.repeat) parts.push(`repeats ${lowerFirst(repeatText(due.repeat))}`);
  return parts.join(', ');
};

export type UpcomingKey = 'overdue' | 'today' | 'tomorrow' | 'week' | 'later' | 'done';
export type UpcomingSection<T> = { key: UpcomingKey; title: string; notes: T[] };

const UPCOMING_ORDER: UpcomingKey[] = ['overdue', 'today', 'tomorrow', 'week', 'later', 'done'];
const UPCOMING_TITLES: Record<UpcomingKey, string> = {
  overdue: 'Overdue',
  today: 'Today',
  tomorrow: 'Tomorrow',
  week: 'Next 7 days',
  later: 'Later',
  done: 'Done',
};

// Placed by date (a snooze doesn't move it); done ones only for 30 days.
export const upcomingKey = (due: Due, now: number): UpcomingKey | null => {
  if (due.doneAt !== null) return due.doneAt >= now - DONE_DAYS * DAY_MS ? 'done' : null;
  if (isOverdue(due, now)) return 'overdue';
  const today = localDate(now);
  if (due.date === today) return 'today';
  if (due.date === addDays(today, 1)) return 'tomorrow';
  return due.date <= addDays(today, 6) ? 'week' : 'later';
};

// No time sorts before a time on the same day ('' < '00:00').
const sortKey = (due: Due) => `${due.date} ${due.time ?? ''}`;
const compare = (a: string, b: string) => (a < b ? -1 : a > b ? 1 : 0);

export const upcomingSections = <T extends { id: string; due: Due | null }>(
  notes: T[],
  now: number,
): UpcomingSection<T>[] => {
  const groups = new Map<UpcomingKey, { note: T; due: Due }[]>();
  for (const note of notes) {
    if (!note.due) continue;
    const key = upcomingKey(note.due, now);
    if (!key) continue;
    groups.set(key, [...(groups.get(key) ?? []), { note, due: note.due }]);
  }
  return UPCOMING_ORDER.flatMap((key) => {
    const list = groups.get(key);
    if (!list) return [];
    list.sort((a, b) =>
      key === 'done'
        ? (b.due.doneAt ?? 0) - (a.due.doneAt ?? 0)
        : compare(sortKey(a.due), sortKey(b.due)) || compare(a.note.id, b.note.id),
    );
    return [{ key, title: UPCOMING_TITLES[key], notes: list.map((entry) => entry.note) }];
  });
};
```

```ts
import { useEffect, useEffectEvent } from 'react';
import { db } from './db';
import { alertTime, formatDue } from './due';
import type { Due, Note } from './types';

export const REMINDER_LOCK = 'noteboard-reminders';
// The clock looks again at least this often, so a laptop waking from sleep
// or a paused phone tab catches up soon after.
export const MAX_WAIT_MS = 60_000;
const MAX_BODY = 120;

export type DueNote = Note & { due: Due };

export const isWatched = (note: Note): note is DueNote =>
  note.due !== null &&
  note.trashedAt === null &&
  !note.archived &&
  note.due.doneAt === null &&
  note.due.alertedAt === null;

export const nextAlertAt = (notes: Note[]): number | null =>
  notes.filter(isWatched).reduce<number | null>((earliest, note) => {
    const at = alertTime(note.due);
    return earliest === null || at < earliest ? at : earliest;
  }, null);

// Marks every reminder whose time has passed as alerted, and returns them.
// One transaction: two tabs can't both take the same reminder.
export const claimDueReminders = (now: number): Promise<DueNote[]> =>
  db.transaction('rw', db.notes, async () => {
    const due = (await db.notes.toArray()).filter(
      (note): note is DueNote => isWatched(note) && alertTime(note.due) <= now,
    );
    const claimed = due.map((note) => ({ ...note, due: { ...note.due, alertedAt: now } }));
    await Promise.all(claimed.map((note) => db.notes.update(note.id, { due: note.due })));
    return claimed;
  });

// Runs the reminder clock while this tab holds the reminders lock (one tab
// at a time; without Web Locks every tab runs it, and claiming still alerts
// each reminder once). `onAlert` gets each batch that came due.
export const useReminderClock = (onAlert: (notes: DueNote[]) => void) => {
  const alert = useEffectEvent(onAlert);

  useEffect(() => {
    let stopped = false;
    let timer: ReturnType<typeof setTimeout> | undefined;
    let running = false;
    let again = false;

    const check = async () => {
      if (running) {
        again = true;
        return;
      }
      running = true;
      try {
        do {
          again = false;
          clearTimeout(timer);
          const due = await claimDueReminders(Date.now()).catch((): DueNote[] => []);
          if (stopped) return;
          if (due.length > 0) alert(due);
          const next = nextAlertAt(await db.notes.toArray().catch((): Note[] => []));
          if (stopped) return;
          const wait = next === null ? MAX_WAIT_MS : Math.min(Math.max(next - Date.now(), 0), MAX_WAIT_MS);
          timer = setTimeout(() => void check(), wait);
        } while (again && !stopped);
      } finally {
        running = false;
      }
    };

    const wake = () => {
      if (document.visibilityState === 'visible') void check();
    };
    let subscription: { unsubscribe: () => void } | undefined;
    const start = () => {
      if (stopped) return;
      // Any change to notes (a date set, snoozed, restored) gets a fresh look.
      subscription = liveQuery(() => db.notes.count()).subscribe({ next: () => void check(), error: () => {} });
      document.addEventListener('visibilitychange', wake);
      window.addEventListener('focus', wake);
    };

    const controller = new AbortController();
    let release: (() => void) | undefined;
    if (typeof navigator !== 'undefined' && navigator.locks) {
      navigator.locks
        .request(REMINDER_LOCK, { signal: controller.signal }, () =>
          new Promise<void>((resolve) => {
            release = resolve;
            start();
          }),
        )
        .catch(() => {});
    } else {
      start();
    }

    return () => {
      stopped = true;
      controller.abort();
      release?.();
      subscription?.unsubscribe();
      clearTimeout(timer);
      document.removeEventListener('visibilitychange', wake);
      window.removeEventListener('focus', wake);
    };
  }, []);
};

const titleOf = (note: Note) => note.title.trim() || 'Untitled note';

export const notificationFor = (notes: DueNote[], now: number) => {
  if (notes.length === 1) {
    const [note] = notes;
    return {
      title: titleOf(note),
      options: {
        body: formatDue(note.due, now),
        tag: `reminder-${note.id}`,
        data: { url: `/note?id=${encodeURIComponent(note.id)}` },
      },
    };
  }
  const titles = notes.map(titleOf).join(', ');
  return {
    title: `${notes.length} reminders`,
    options: {
      body: titles.length > MAX_BODY ? `${titles.slice(0, MAX_BODY - 1)}…` : titles,
      tag: 'reminders',
      data: { url: '/?view=upcoming' },
    },
  };
};

// Only while the page is out of view: the reminder panel covers a visible
// page. Through the service worker when there is one (Android requires it);
// otherwise (development) a plain Notification whose click calls `onClick`.
export const showSystemNotification = async (notes: DueNote[], now: number, onClick: (url: string) => void) => {
  if (typeof Notification === 'undefined' || Notification.permission !== 'granted') return;
  if (document.visibilityState === 'visible') return;
  const { title, options } = notificationFor(notes, now);
  try {
    const registration = await navigator.serviceWorker?.getRegistration();
    if (registration?.active) {
      await registration.showNotification(title, options);
      return;
    }
    const notification = new Notification(title, options);
    notification.onclick = () => {
      window.focus();
      onClick(options.data.url);
      notification.close();
    };
  } catch {
    // The reminder panel still shows it.
  }
};

export const notificationsBlocked = () => typeof Notification === 'undefined' || Notification.permission === 'denied';

// Must be called straight from a click: browsers only prompt then.
export const requestNotificationPermission = () => {
  if (typeof Notification === 'undefined' || Notification.permission !== 'default') return;
  Promise.resolve(Notification.requestPermission()).catch(() => {});
};
```

```js
// A reminder notification was clicked: bring Noteboard forward and open the
// note (or Upcoming). An open window navigates itself, keeping its state;

export const openFromNotification = async (clients, origin, url) => {
  const windows = await clients.matchAll({ type: 'window', includeUncontrolled: true });
  const open = windows.find((client) => new URL(client.url).origin === origin);
  if (open) {
    await open.focus();
    open.postMessage({ type: 'open-reminder', url });
    return;
  }
  await clients.openWindow(new URL(url, origin).href);
```

```ts
import { addDays, isDateString, isTimeString, localDate, MONTH_SHORT, parseDate, WEEKDAY_SHORT, weekdayOf } from './calendar';
import { formatTime } from './due';
import { presetOf, presetRepeat, type RepeatPreset } from './repeat';
import type { Due, Repeat, RepeatEnd, RepeatUnit } from './types';

export type RepeatChoice = 'none' | RepeatPreset | 'custom';

// The date popover's fields, as the inputs hold them (strings).
export type DueForm = {
  date: string;
  time: string; // '' = no time
  repeat: RepeatChoice;
  every: string;
  unit: RepeatUnit;
  weekdays: number[];
  ends: 'never' | 'on' | 'after';
  endDate: string;
  endCount: string;
};

export type DueFormErrors = Partial<Record<'date' | 'every' | 'endDate' | 'endCount', string>>;

export const formFromDue = (due: Due | null, today: string): DueForm => {
  const repeat = due?.repeat ?? null;
  const date = due?.date ?? today;
  return {
    date,
    time: due?.time ?? '',
    repeat: repeat ? presetOf(repeat) : 'none',
    every: String(repeat?.every ?? 1),
    unit: repeat?.unit ?? 'week',
    // Custom weeks start on the date's own weekday.
    weekdays: repeat && repeat.weekdays.length > 0 ? repeat.weekdays : [weekdayOf(date)],
    ends: repeat?.ends.type ?? 'never',
    endDate: repeat?.ends.type === 'on' ? repeat.ends.date : '',
    endCount: repeat?.ends.type === 'after' ? String(repeat.ends.count) : '5',
  };
};

const wholeIn = (text: string, min: number, max: number) => {
  const value = Number(text);
  return text.trim() !== '' && Number.isInteger(value) && value >= min && value <= max ? value : null;
};

export const readDueForm = (
  form: DueForm,
): { ok: true; date: string; time: string | null; repeat: Repeat | null } | { ok: false; errors: DueFormErrors } => {
  const errors: DueFormErrors = {};
  const dateOk = isDateString(form.date);
  if (!dateOk) errors.date = 'Choose a date.';
  const time = isTimeString(form.time) ? form.time : null;
  let repeat: Repeat | null = null;
  if (form.repeat !== 'none' && form.repeat !== 'custom' && dateOk) repeat = presetRepeat(form.repeat, form.date);
  if (form.repeat === 'custom') {
    const every = wholeIn(form.every, 1, 99);
    if (every === null) errors.every = 'Enter a number from 1 to 99.';
    let ends: RepeatEnd = { type: 'never' };
    if (form.ends === 'on') {
      if (!isDateString(form.endDate)) errors.endDate = 'Choose an end date.';
      else if (dateOk && form.endDate <= form.date) errors.endDate = 'The end date must be after the date.';
      else ends = { type: 'on', date: form.endDate };
    }
    if (form.ends === 'after') {
      const count = wholeIn(form.endCount, 1, 999);
      if (count === null) errors.endCount = 'Enter a number from 1 to 999.';
      else ends = { type: 'after', count, done: 0 };
    }
    if (every !== null && dateOk) {
      const weekdays = form.unit === 'week' ? [...new Set(form.weekdays)].sort((a, b) => a - b) : [];
      repeat = { every, unit: form.unit, weekdays, anchor: form.date, ends };
    }
  }
  if (Object.keys(errors).length > 0) return { ok: false, errors };
  return { ok: true, date: form.date, time, repeat };
};

export type QuickPick = { label: string; detail: string; date: string; time: string };

export const quickPicks = (now: number, locale?: string): QuickPick[] => {
  const today = localDate(now);
  const picks: QuickPick[] = [];
  if (new Date(now).getHours() < 17) {
    picks.push({ label: 'Later today', detail: formatTime('18:00', locale), date: today, time: '18:00' });
  }
  picks.push({ label: 'Tomorrow', detail: formatTime('09:00', locale), date: addDays(today, 1), time: '09:00' });
  const toMonday = (8 - weekdayOf(today)) % 7 || 7;
  picks.push({ label: 'Next week', detail: `Mon ${formatTime('09:00', locale)}`, date: addDays(today, toMonday), time: '09:00' });
  return picks;
};

export const repeatOptions = (date: string): { value: RepeatChoice; label: string }[] => {
  const valid = isDateString(date);
  const { month, day } = valid ? parseDate(date) : { month: 1, day: 1 };
  return [
    { value: 'none', label: 'Doesn’t repeat' },
    { value: 'daily', label: 'Daily' },
    { value: 'weekdays', label: 'Weekdays (Mon–Fri)' },
    { value: 'weekly', label: valid ? `Weekly on ${WEEKDAY_SHORT[weekdayOf(date)]}` : 'Weekly' },
    { value: 'monthly', label: valid ? `Monthly on day ${day}` : 'Monthly' },
    { value: 'yearly', label: valid ? `Yearly on ${day} ${MONTH_SHORT[month - 1]}` : 'Yearly' },
    { value: 'custom', label: 'Custom…' },
  ];
};
```

```tsx
const DueChip = ({ due, readOnly, onEdit, onRemove, onMarkDone, onReopen }: Props) => {

  const done = isDone(due);
  const state = done ? 'done' : isOverdue(due, now) ? 'overdue' : undefined;

      <div className={styles.row}>

          {content}
        </p>
      </div>
    );
  }

    <div className={styles.row}>

        {content}
```

```tsx
    draftRef.current = { ...draftRef.current, items };
```

```tsx
  note.due.doneAt === null &&
  note.trashedAt === null &&
  !note.archived;

type RowProps = {
  note: DueNote;
  now: number;
  onOpen: () => void;
  onSnooze: (until: number) => void;
  onDone: () => void;
  onDismiss: () => void;
};

const ReminderRow = ({ note, now, onOpen, onSnooze, onDone, onDismiss }: RowProps) => {
  const title = titleOf(note);
  const tomorrow = () => localMoment(addDays(localDate(Date.now()), 1), '09:00');
  return (
    <li className={styles.row}>
      <AlarmClock size={18} aria-hidden className={styles.icon} />
      <div className={styles.text}>
        <p className={styles.title}>{title}</p>
        <p className={styles.when}>{formatDue(note.due, now)}</p>
      </div>
      <div className={styles.actions}>
        <button type='button' className={styles.button} onClick={onOpen}>
          Open
        </button>
        <Menu
          align='start'
          items={[
            { label: '10 minutes', onSelect: () => onSnooze(Date.now() + 10 * MINUTE) },
            { label: '1 hour', onSelect: () => onSnooze(Date.now() + 60 * MINUTE) },
            { label: `Tomorrow, ${formatTime('09:00')}`, onSelect: () => onSnooze(tomorrow()) },
          ]}
          trigger={
            <button type='button' className={styles.button}>
              Snooze
            </button>
          }
        />
        <button type='button' className={styles.button} onClick={onDone}>
          Mark done
        </button>
        <IconButton label={`Dismiss reminder for ${title}`} onClick={onDismiss}>
          <X size={16} aria-hidden />
        </IconButton>
      </div>
    </li>
  );
};

// Reminders that came due in this tab, newest first. Rows stay until acted
// on (WCAG 2.2.1); they aren't kept across reloads (the note stays overdue).
const ReminderAlerts = () => {
  const router = useRouter();
  const toast = useToast();
  const now = useNow();
  const [rows, setRows] = useState<string[]>([]);
  const [announcement, setAnnouncement] = useState('');

  useReminderClock((due) => {
    const at = Date.now();
    void showSystemNotification(due, at, (url) => void router.push(url));
    const ids = due.map((note) => note.id);
    setRows((current) => [...ids, ...current.filter((id) => !ids.includes(id))]);
    setAnnouncement(
      due.length === 1 ? `Reminder: ${titleOf(due[0])}, ${formatDue(due[0].due, at)}` : `${due.length} reminders`,
    );
  });

  // A click on a system notification, relayed by the service worker.
  useEffect(() => {
    const container = typeof navigator !== 'undefined' ? navigator.serviceWorker : undefined;
    if (!container) return;
    const onMessage = (event: MessageEvent) => {
      const data = event.data as { type?: unknown; url?: unknown } | null;
      if (data?.type === 'open-reminder' && typeof data.url === 'string' && data.url.startsWith('/')) {
        void router.push(data.url);
      }
    };
    container.addEventListener('message', onMessage);
    return () => container.removeEventListener('message', onMessage);
  }, [router]);

  const key = rows.join('\n');
  const notes = useLiveQuery(() => (key ? db.notes.bulkGet(key.split('\n')) : Promise.resolve([])).catch(() => []), [key]);
  const byId = new Map((notes ?? []).flatMap((note) => (note ? [[note.id, note] as const] : [])));
  const active = rows.map((id) => byId.get(id)).filter(isAlerting);
  const shown = active.slice(0, MAX_ROWS);
  const more = active.length - shown.length;

  const dismiss = (id: string) => setRows((current) => current.filter((row) => row !== id));

  const open = (id: string) => {
    dismiss(id);
    openEditor(router, noteHref(id));
  };

  const done = async (id: string) => {
    const before = await markDone(id);
    if (!before) return;
    const after = await db.notes.get(id);
    const movedTo = after?.due && after.due.doneAt === null ? formatDue(after.due, Date.now()) : null;
    toast({
      message: movedTo ? `Moved to ${movedTo}` : 'Marked done',
      actionLabel: 'Undo',
      onAction: () => void putBackDue(id, before.due, before.items),
    });
  };

  return (
    <>
      <div role='status' aria-live='polite' className={styles.srOnly}>
        {announcement}
      </div>
      {shown.length > 0 && (
        <section className={styles.panel} aria-label='Reminders'>
          <ul className={styles.list}>
            {shown.map((note) => (
              <ReminderRow
                key={note.id}
                note={note}
                now={now}
                onOpen={() => open(note.id)}
                onSnooze={(until) => void snooze(note.id, until)}
                onDone={() => void done(note.id)}
                onDismiss={() => dismiss(note.id)}
              />
            ))}
          </ul>
          {more > 0 && (
            <Link href='/?view=upcoming' className={styles.more}>
              {`+${more} more · View in Upcoming`}
            </Link>
          )}
        </section>
      )}
    </>
  );
};

export default ReminderAlerts;
```
