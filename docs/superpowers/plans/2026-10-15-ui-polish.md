`lib/tasks.ts`:

```ts
// Tick-box lines in a text note's Markdown ("- [ ] milk"), so a card can tick
// one without opening the editor. Matches what marked treats as a task item:
// a bullet or number (inside any quotes), then [ ], [x] or [X], then a space.
const TASK = /^((?:\s*>)*\s*(?:[-+*]|\d+[.)])\s+)\[([ xX])\](?=\s|$)/;
const FENCE = /^\s*(```|~~~)/;

export const taskLines = (markdown: string): number[] => {
  const lines: number[] = [];
  let fence: string | null = null;
  markdown.split('\n').forEach((line, index) => {
    const opener = FENCE.exec(line)?.[1];
    if (opener) {
      if (fence === null) fence = opener;
      else if (opener === fence) fence = null;
      return;
    }
    if (fence === null && TASK.test(line)) lines.push(index);
  });
  return lines;
};

export const toggleTask = (markdown: string, index: number): string => {
  const line = taskLines(markdown)[index];
  if (line === undefined) return markdown;
  const lines = markdown.split('\n');
  lines[line] = lines[line].replace(TASK, (_all, prefix: string, mark: string) => `${prefix}[${mark === ' ' ? 'x' : ' '}]`);
  return lines.join('\n');
};
```

```tsx
type RenderOptions = { onToggleTask?: (index: number) => void; highlight?: string };

const renderer = ({ onToggleTask }: RenderOptions) => {
  let task = 0; // tick boxes are numbered in document order, as lib/tasks counts them

  const text = (value: string): ReactNode => value;

  const inline = (tokens: Token[] | undefined): ReactNode =>
    (tokens ?? []).map((token, index) => <Fragment key={index}>{inlineToken(token)}</Fragment>);

  const inlineToken = (token: Token): ReactNode => {
    switch (token.type) {
      case 'strong':
        return <strong>{inline(token.tokens)}</strong>;
      case 'em':
        return <em>{inline(token.tokens)}</em>;
      case 'del':
        return <s>{inline(token.tokens)}</s>;
      case 'codespan':
        return <code>{token.text}</code>;
      case 'br':
        return <br />;
      // A click anywhere on a card opens the note, so links are shown, not followed.
      case 'link':
        return <span className={styles.link}>{inline(token.tokens)}</span>;
      case 'escape':
        return token.text;
      case 'text':
        return 'tokens' in token && token.tokens ? inline(token.tokens) : text(textOf(token));
      default:
        return 'tokens' in token && token.tokens ? inline(token.tokens) : text(textOf(token));
    }
  };

  const blocks = (tokens: Token[]): ReactNode =>
    tokens.map((token, index) => <Fragment key={index}>{block(token)}</Fragment>);

  const listItem = (item: Tokens.ListItem, index: number) => {
    if (!item.task) return <li key={index}>{blocks(item.tokens)}</li>;
    const number = task++;
    const label = item.text.replace(/^\s*\[[ xX]\]\s*/, '').split('\n')[0] || 'item';
    // marked puts the checkbox first in the item's tokens; draw our own instead.
    const rest = item.tokens.filter((token) => token.type !== 'checkbox');
    return (
      <li key={index} className={styles.task} data-checked={item.checked ? 'true' : undefined}>
        <input
          type='checkbox'
          checked={!!item.checked}
          disabled={!onToggleTask}
          aria-label={`Mark ‘${label}’ ${item.checked ? 'not done' : 'done'}`}
          onChange={() => onToggleTask?.(number)}
        />
        <span>{blocks(rest)}</span>
      </li>
    );
  };

  const block = (token: Token): ReactNode => {
    switch (token.type) {
      case 'space':
      case 'def':
        return null;
      // Not real headings: the card title is the heading.
      case 'heading':
        return (
          <p className={styles.heading} data-level={token.depth}>
            {inline(token.tokens)}
          </p>
        );
      case 'paragraph':
        return <p>{inline(token.tokens)}</p>;
      case 'text':
        return 'tokens' in token && token.tokens ? inline(token.tokens) : text(textOf(token));
      case 'list': {
        const items = (token as Tokens.List).items.map(listItem);
        const tasks = (token as Tokens.List).items.some((item) => item.task);
        return token.ordered ? (
          <ol start={typeof token.start === 'number' ? token.start : undefined}>{items}</ol>
        ) : (
          <ul className={tasks ? styles.tasks : undefined}>{items}</ul>
        );
      }
      case 'blockquote':
        return <blockquote>{blocks(token.tokens ?? [])}</blockquote>;
      case 'code':
        return (
          <pre>
            <code>{token.text}</code>
          </pre>
        );
      case 'hr':
        return <hr />;
      default:
        return <p>{decodeEntities(token.raw)}</p>;
    }
  };

  return { blocks };
};
```

`lib/highlight.ts`:

```ts
// The query's matches in a piece of text, for <mark> on search results.
export const splitMatches = (text: string, query: string): { text: string; match: boolean }[] => {
  const needle = query.trim().toLowerCase();
  if (!needle) return [{ text, match: false }];
  const parts: { text: string; match: boolean }[] = [];
  const lower = text.toLowerCase();
  let from = 0;
  for (let at = lower.indexOf(needle); at !== -1; at = lower.indexOf(needle, at + needle.length)) {
    if (at > from) parts.push({ text: text.slice(from, at), match: false });
    parts.push({ text: text.slice(at, at + needle.length), match: true });
    from = at + needle.length;
  }
  if (from < text.length) parts.push({ text: text.slice(from), match: false });
  return parts;
};
```

import { useSyncExternalStore } from 'react';

// A media query as state; `serverValue` is used for the server render.
export const useMediaQuery = (query: string, serverValue = false): boolean =>
  useSyncExternalStore(
    (onChange) => {
      const list = window.matchMedia(query);
      list.addEventListener('change', onChange);
      return () => list.removeEventListener('change', onChange);
    },
    () => window.matchMedia(query).matches,
    () => serverValue,
  );

```tsx
import { Dialog } from 'radix-ui';
import { useRef, useState, useSyncExternalStore, type ReactNode } from 'react';
import styles from './AppShell.module.css';
import Sidebar from './Sidebar';

const KEY = 'sidebar';
const DESKTOP = '(min-width: 1024px)';

const listeners = new Set<() => void>();
const readRail = () => {
  try {
    return localStorage.getItem(KEY) === 'rail';
  } catch {
    return false;
  }
};
const writeRail = (rail: boolean) => {
  try {
    localStorage.setItem(KEY, rail ? 'rail' : 'open');
  } catch {
    // Only a preference.
  }
  listeners.forEach((listener) => listener());
};

type Props = {
  renderTop: (onMenu: () => void) => ReactNode;
  onEditLabels: () => void;
  children: ReactNode;
};

// Header, sidebar and content. Widths decide the layout in CSS (so the first
// paint is right); ☰ collapses the sidebar on desktop and opens a drawer below it.
const AppShell = ({ renderTop, onEditLabels, children }: Props) => {
  const rail = useSyncExternalStore(
    (onChange) => {
      listeners.add(onChange);
      return () => listeners.delete(onChange);
    },
    readRail,
    () => false,
  );
  const [drawer, setDrawer] = useState(false);
  const menuButton = useRef<HTMLElement | null>(null);

  const onMenu = () => {
    menuButton.current = document.activeElement as HTMLElement | null;
    if (window.matchMedia(DESKTOP).matches) writeRail(!rail);
    else setDrawer(true);
  };

  return (
    <div className={styles.shell} data-sidebar={rail ? 'rail' : 'open'}>
      {renderTop(onMenu)}
      <div className={styles.body}>
        <aside className={styles.side}>
          <Sidebar variant='side' onEditLabels={onEditLabels} />
        </aside>
        <div className={styles.content}>{children}</div>
      </div>
      <Dialog.Root open={drawer} onOpenChange={setDrawer}>
        <Dialog.Portal>
          <Dialog.Overlay className={styles.scrim} />
          <Dialog.Content
            className={styles.drawer}
            aria-describedby={undefined}
            onCloseAutoFocus={(event) => {
              event.preventDefault();
              menuButton.current?.focus();
            }}
          >
            <Dialog.Title className={styles.drawerTitle}>Sections</Dialog.Title>
            <Sidebar
              variant='drawer'
              onNavigate={() => setDrawer(false)}
              onEditLabels={() => {
                setDrawer(false);
                onEditLabels();
              }}
            />
          </Dialog.Content>
        </Dialog.Portal>
      </Dialog.Root>
    </div>
  );
};

export default AppShell;

import { RadioGroup } from 'radix-ui';
import { viewTitle } from '../../lib/filters';
import type { BoardFilter, BoardView } from '../../lib/types';
import styles from './ViewHeader.module.css';

type Props = {
  filter: BoardFilter;
  labelName?: string;
  total: number;
  onView: (view: BoardView) => void;
  onEmptyTrash: () => void;
};

const KINDS: { view: BoardView; label: string }[] = [
  { view: 'all', label: 'All' },
  { view: 'notes', label: 'Text' },
  { view: 'lists', label: 'Lists' },
];

// The page's heading: the section's name, or the search and its count.
const ViewHeader = ({ filter, labelName, total, onView, onEmptyTrash }: Props) => {
  const query = filter.query.trim();
  const onNotes = !query && filter.labelId === null && (filter.view === 'all' || filter.view === 'notes' || filter.view === 'lists');
  return (
    <div className={styles.header}>
      <div className={styles.row}>
        <h1 className={styles.title}>{query ? `Results for “${query}”` : viewTitle(filter, labelName)}</h1>
        {query && (
          <p role='status' className={styles.count}>
            {total === 1 ? '1 note' : `${total} notes`}
          </p>
        )}
        {onNotes && (
          <RadioGroup.Root
            className={styles.switch}
            aria-label='Show'
            value={filter.view}
            onValueChange={(view) => onView(view as BoardView)}
            orientation='horizontal'
          >
            {KINDS.map(({ view, label }) => (
              <RadioGroup.Item key={view} value={view} className={styles.option}>
                {label}
              </RadioGroup.Item>
            ))}
          </RadioGroup.Root>
        )}
      </div>
      {filter.view === 'trash' && total > 0 && (
        <div className={styles.trash}>
          <p>Notes in Trash are deleted after 30 days.</p>
          <button type='button' className={styles.textButton} onClick={onEmptyTrash}>

import { useSyncExternalStore } from 'react';

const inset = () => {
  const view = window.visualViewport;
  return view ? Math.max(0, Math.round(window.innerHeight - view.height - view.offsetTop)) : 0;
};

// How much of the bottom of the window the on-screen keyboard covers.
export const useKeyboardInset = (): number =>
  useSyncExternalStore(
    (onChange) => {
      const view = window.visualViewport;
      view?.addEventListener('resize', onChange);
      view?.addEventListener('scroll', onChange);
      return () => {
        view?.removeEventListener('resize', onChange);
        view?.removeEventListener('scroll', onChange);
      };
    },
    inset,
    () => 0,
  );

  const [r, g, b] = rgb(hex);
  const max = Math.max(r, g, b);
  const min = Math.min(r, g, b);
  const l = (max + min) / 2;
  const s = max === min ? 0 : (max - min) / (1 - Math.abs(2 * l - 1));
  const h = max === min ? 0 : max === r ? 60 * (((g - b) / (max - min)) % 6) : max === g ? 60 * ((b - r) / (max - min) + 2) : 60 * ((r - g) / (max - min) + 4);
  return { h: (h + 360) % 360, s, l };
};

const COLORS = ['yellow', 'green', 'blue', 'pink', 'purple', 'grey'];

describe('note colours', () => {
  it('keep body and title text at AA contrast in both themes', () => {
    const lightText = value("[data-theme='light']", '--text') || value(':root,\n[data-theme=\'light\']', '--text');
    const darkText = value("[data-theme='dark']", '--text');
    for (const color of COLORS) {
      const light = `[data-color='${color}']`;
      const dark = `[data-theme='dark'] [data-color='${color}']`;
      expect(contrast(value(light, '--note-ink'), value(light, '--note-bg'))).toBeGreaterThanOrEqual(4.5);
      expect(contrast(value(dark, '--note-ink'), value(dark, '--note-bg'))).toBeGreaterThanOrEqual(4.5);
      if (lightText) expect(contrast(lightText, value(light, '--note-bg'))).toBeGreaterThanOrEqual(4.5);
      expect(contrast(darkText, value(dark, '--note-bg'))).toBeGreaterThanOrEqual(4.5);
    }
  });

  it('makes dark yellow read as amber, not olive', () => {
    const { h, s } = hsl(value("[data-theme='dark'] [data-color='yellow']", '--note-bg'));
    expect(h).toBeGreaterThanOrEqual(30);
    expect(h).toBeLessThanOrEqual(45);
    expect(s).toBeGreaterThanOrEqual(0.5);
  });
});
